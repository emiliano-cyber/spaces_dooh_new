#  PASO 3 · ESCRIBIRLAS EN LA INSTANCIA   🖥️ EN SU DROPLET
##############################################################
#
#  Mira que hay (imprime CLAVES, no valores):
#
    sed 's/=.*/=<valor>/' /etc/space-os/instancia.env
#
#  Respaldo del archivo y edicion:
#
    cp /etc/space-os/instancia.env /etc/space-os/instancia.env.bak
    nano /etc/space-os/instancia.env
#
#  >>> LAS CUATRO PRIMERAS YA EXISTEN EN EL ARCHIVO, vacias, bajo
#  >>> "Respaldos y registro de corridas". EDITALAS, no añadas
#  >>> lineas nuevas: el archivo se sourcea y una clave duplicada
#  >>> hace que gane la ultima.
#  >>> SPACES_REGION **no existe** en la plantilla: esa si se añade.
#
#    SPACES_KEY="<KEY>"
#    SPACES_SECRET="<SECRET>"
#    SPACES_BUCKET="space-os-respaldos"
#    SPACES_REGION="<la del PASO 1>"      # <-- SE AÑADE
#    LOGS_BUCKET="space-os-logs"
#
#  >>> LAS COMILLAS NO SON DECORACION. `update.sh:700` hace
#  >>> `. "$CONF"`: bash SOURCEA este archivo. Un valor con un
#  >>> espacio sin comillas hace que bash EJECUTE la segunda
#  >>> palabra COMO ROOT, cada noche. Medido el 2026-09-02
#  >>> (expediente de F3.5, punto 12).
#
    chmod 600 /etc/space-os/instancia.env
    ls -l /etc/space-os/instancia.env      # -rw-------
#
#  Comprueba que quedaron puestas. Imprime LONGITUDES, nunca
#  valores, asi que se puede pegar en un chat sin riesgo:
#
    bash -c 'set -a; . /etc/space-os/instancia.env; set +a; for v in SPACES_KEY SPACES_SECRET SPACES_BUCKET SPACES_REGION LOGS_BUCKET; do val="${!v}"; printf "%-15s %s\n" "$v" "$([ -n "$val" ] && echo "puesta, ${#val} caracteres" || echo VACIA)"; done'
