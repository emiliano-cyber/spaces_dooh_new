# ############################################################
#  2 · La clave SSH del PADRE, EN la cuenta de DO
# ############################################################
#  `--crear-droplet` crea la maquina, pero el script entra luego por
#  `ssh root@HOST`. Si la clave no esta en la cuenta ANTES, el droplet
#  nace sin ella, el script muere en el primer ssh -- y la maquina se
#  queda creada y pagandose.

ls -la /root/.ssh/id_*.pub

#  ESPERADO: al menos un .pub (normalmente id_ed25519.pub).
#  Si no hay ninguno: ssh-keygen -t ed25519 -C "padre"

doctl compute ssh-key list

#  Si no hay ninguna, generala y subela. DOS TRAMPAS MEDIDAS EL
#  2026-09-03, las dos en este paso:
#
#  TRAMPA 1 · `--public-key-file` FALLA con doctl instalado como
#  snap, incluso siendo root:
#      Error: open /root/.ssh/id_ed25519.pub: permission denied
#  Un snap esta confinado y su interfaz `home` NO cubre /root. No es
#  un problema de permisos del archivo. La salida es que lea el
#  SHELL, no doctl:

ssh-keygen -t ed25519 -C "padre" -f /root/.ssh/id_ed25519 -N ""
doctl compute ssh-key create padre --public-key "$(cat /root/.ssh/id_ed25519.pub)"

#  (Alternativa, si se prefiere el import: `snap connect doctl:ssh-keys`
#   abre esa interfaz. La de arriba no necesita tocar el confinamiento.)
#
#  TRAMPA 2 · DigitalOcean muestra el fingerprint en MD5 y
#  `ssh-keygen -lf` lo da en SHA256, asi que NO se parecen y es facil
#  concluir que subiste otra clave. Para comparar de verdad:

doctl compute ssh-key list
ssh-keygen -E md5 -lf /root/.ssh/id_ed25519.pub

#  ESPERADO: el hexadecimal de la segunda salida (sin el prefijo
#  `MD5:`) es EXACTAMENTE la columna FingerPrint de la primera.
