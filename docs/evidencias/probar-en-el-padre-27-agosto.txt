# OJO CON `sudo` Y LAS VARIABLES DE ENTORNO
# `DATABASE_URL=... sudo ... node` NO FUNCIONA: pone la
# variable en el entorno de sudo, y sudo lo limpia antes de
# lanzar node, que la recibe vacia y aborta pidiendola. La
# variable va AL OTRO LADO de sudo, con `env`. Se usa `env` y
# no `sudo -u postgres VAR=... node` porque eso depende de que
# `setenv` este permitido en el sudoers; `env` es un binario y
# funciona siempre.
# (La tarjeta del despliegue del 27 traia la version rota. No
#  se noto porque su bloque de migracion nunca se corrio.)

# NOTA DE ALCANCE, dicha antes de empezar: la instancia que
# vamos a mirar es la app del PROPIO PADRE en space-os.io. No
# es un owner remoto -- eso necesita el registry. Pero SI es
# una peticion HTTPS real, por nginx, a un Next real con su
# base: la logica del panel se ejerce entera. Lo unico que no
# demuestra es la distancia.
#
# DEMO (el 3001) no sirve para esto todavia: no tiene dominio
# desde el ADR 0021 y su certificado nunca se emitio.

# ------------------------------------------------------------
# F1 [PADRE] - Medir antes de tocar: donde esta el entorno.
# ------------------------------------------------------------

cut -d= -f1 /var/www/Spaces/apps/web/.env.production | sort; echo "[fin]"

# Si entre el comando y [fin] no sale nada, las tres variables
# faltan -- que es lo esperado. `esElPanel()` devuelve false sin
# FLOTA_TOKEN configurado: ausente = cerrado, a proposito.

# F2 [PADRE] - Respaldo del .env antes de anadirle nada.

cp /var/www/Spaces/apps/web/.env.production /root/env.padre.bak.$(date +%F_%H%M%S) && ls -l /root/env.padre.bak.*

# F3 [PADRE] - Generar un token de flota y anadir las tres
#              variables. El token se imprime: GUARDALO.

cd /var/www/Spaces && T=$(openssl rand -hex 32) && printf '\nFLOTA_TOKEN=%s\nSPACE_OS_VERSION=v0.1.0-padre\nCANAL=beta\n' "$T" >> apps/web/.env.production && echo "TOKEN: $T"

# F4 [PADRE] - Reiniciar. SPACE_OS_VERSION se lee en tiempo de
#              PETICION (force-dynamic), pero el proceso tiene
#              que releer el .env: hay que reiniciarlo.

pm2 restart spaces-web && sleep 3 && pm2 list

# F5 [PADRE] - La ruta contesta lo de siempre SIN token.

curl -s https://space-os.io/spaces-dooh/api/version/; echo

# Esperado exacto:  {"ok":true}
# Nada mas. Sin token no se dice la version -- publicarla le
# ahorra el trabajo a quien busca una vulnerabilidad conocida.

# F6 [PADRE] - Y CON token dice lo que el panel necesita.
#              El token se lee del archivo: no hace falta
#              pegarlo, ni que salga de la maquina.

T=$(grep '^FLOTA_TOKEN=' /var/www/Spaces/apps/web/.env.production | cut -d= -f2); curl -s -H "x-flota-token: $T" https://space-os.io/spaces-dooh/api/version/; echo

# Esperado: {"ok":true,"version":"v0.1.0-padre","ultimaMigracion":"...","base":"ok","canal":"beta","uptime":N}
#
# SEIS CLAVES Y NI UNA MAS. Ni cuantas organizaciones hay, ni
# como se llaman, ni una cifra del negocio. Si ves cualquier
# otra clave, PARA y pasamelo: es un fallo de la promesa
# comercial, no un detalle.

# F7 [PADRE] - Montar el inventario del panel.
#              flota.json NO esta en git a proposito: seria la
#              lista de clientes con sus dominios dentro del
#              repositorio.
#
# Pega este bloque ENTERO de una vez.

cd /var/www/Spaces/apps/flota && cat > flota.json <<'JSON'
{
  "canales": { "estable": "v0.1.0-padre", "beta": "v0.1.0-padre" },
  "instancias": [
    { "nombre": "padre",      "dominio": "space-os.io",               "canal": "beta" },
    { "nombre": "inventario", "dominio": "inventario.ejemplo.invalid", "canal": "estable" },
    { "nombre": "vallas",     "dominio": "vallas.ejemplo.invalid",     "canal": "estable" }

}
JSON

# F8 [PADRE] - ESTADO 1 de 3: `al-dia`.
#              El token va por entorno y NUNCA en el inventario.
#              El nombre en mayusculas: padre -> FLOTA_TOKEN_PADRE

cd /var/www/Spaces/apps/flota && export FLOTA_TOKEN_PADRE=$(grep '^FLOTA_TOKEN=' /var/www/Spaces/apps/web/.env.production | cut -d= -f2) && node estado.mjs; echo "codigo: [$?]"

# ESPERADO:
#   padre       al-dia          (corre justo la version del canal)
#   inventario  sin-respuesta   (dominio .invalid, no existe)
#   vallas      sin-respuesta
#   salida: 0
#
# LOS DOS `sin-respuesta` SON PARTE DE LA PRUEBA, no un fallo:
# un panel que revienta cuando una instancia se cae no sirve
# para vigilar, porque el dia que hace falta es el dia que no
# arranca. Por eso se comprueba `salida: 0`.

# F9 [PADRE] - ESTADO 2 de 3: `rezagada`.
#              Se mueve el canal, no la instancia.

cd /var/www/Spaces/apps/flota && sed -i 's/"beta": "v0.1.0-padre"/"beta": "v0.2.0-padre"/' flota.json && node estado.mjs; echo "codigo: [$?]"
