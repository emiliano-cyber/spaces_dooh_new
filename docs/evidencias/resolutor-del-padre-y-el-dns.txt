import { readFileSync } from 'node:fs'
const env = Object.fromEntries(
  readFileSync('/etc/space-os/ejecutor.env', 'utf8')
    .split('\n')
    .filter((l) => l.includes('=') && !l.trim().startsWith('#'))
    .map((l) => [l.slice(0, l.indexOf('=')).trim(), l.slice(l.indexOf('=') + 1).trim()]),
)
const zonas = JSON.parse(env.CLOUDFLARE_ZONAS ?? '{}')
const zona = zonas['space-os.io']
if (!zona) { console.error('no hay zona space-os.io en CLOUDFLARE_ZONAS'); process.exit(1) }
const r = await fetch(
  `https://api.cloudflare.com/client/v4/zones/${zona}/dns_records?type=A&per_page=100`,
  { headers: { Authorization: `Bearer ${env.CLOUDFLARE_TOKEN}` } },
)
const d = await r.json()
if (!d.success) { console.error('Cloudflare:', d.errors?.map((e) => e.message).join('; ')); process.exit(1) }
for (const x of d.result) {
  if (x.name.includes('prueba')) console.log(x.name, '->', x.content, 'proxied=' + x.proxied)
}
console.log('--- total de registros A en la zona:', d.result.length)
FIN
