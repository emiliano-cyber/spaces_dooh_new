# ############################################################
#  P0 - ANTES DE TOCAR NADA: el censo de RFC
# ############################################################
# La validacion nueva del RFC rechaza fechas imposibles (mes 13,
# 31 de febrero). Esto busca los que YA ESTAN GUARDADOS y que la
# regla nueva rechazaria: no rompe nada de lo existente, pero a
# esos registros no se les podra guardar una edicion hasta
# corregir el dato.
#
# Es de SOLO LECTURA. Pasame el resultado antes de seguir.
#
# NOTA: la consulta usa `not (... ~ ...)` y no el operador `!~`, que es lo
# natural en SQL. El motivo no es de SQL sino de bash: el `!` dispara la
# EXPANSION DE HISTORIAL aunque este dentro de comillas dobles, y la linea
# muere con "-bash: !~: event not found" sin llegar a Postgres. Medido el
# 27/08 en esta misma consola.

sudo -u postgres psql -d spaces_prod -c "with rfcs as (select 'tenants' t, id::text i, rfc from tenants where coalesce(btrim(rfc),'') <> '' union all select 'clientes', id::text, rfc from clientes where coalesce(btrim(rfc),'') <> '' union all select 'arrendadores', id::text, rfc from arrendadores where coalesce(btrim(rfc),'') <> ''), ok as (select t,i,rfc,upper(btrim(rfc)) r from rfcs where upper(btrim(rfc)) ~ ('^[A-Z' || chr(209) || '&]{3,4}[0-9]{6}[A-Z0-9]{3}$')) select t,i,rfc,'forma' motivo from rfcs where not (upper(btrim(rfc)) ~ ('^[A-Z' || chr(209) || '&]{3,4}[0-9]{6}[A-Z0-9]{3}$')) union all select t,i,rfc,'fecha imposible' from ok where substring(r from ('^[A-Z' || chr(209) || '&]{3,4}[0-9]{2}([0-9]{2})'))::int not between 1 and 12 or substring(r from ('^[A-Z' || chr(209) || '&]{3,4}[0-9]{4}([0-9]{2})'))::int > (array[0,31,29,31,30,31,30,31,31,30,31,30,31])[substring(r from ('^[A-Z' || chr(209) || '&]{3,4}[0-9]{2}([0-9]{2})'))::int];"
