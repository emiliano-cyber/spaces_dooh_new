#  Las 39 tablas son de `postgres`. Una migracion que haga

echo ok; CLAVE_MIG=$(openssl rand -hex 24); echo "$CLAVE_MIG" > /root/.clave-migrador-demo; chmod 600 /root/.clave-migrador-demo; echo "clave generada y guardada en /root/.clave-migrador-demo"

echo ok; sudo -u postgres psql -v ON_ERROR_STOP=1 -c "create role spaces_migrador login password '$CLAVE_MIG' nosuperuser nocreaterole noinherit"

echo ok; sudo -u postgres psql -v ON_ERROR_STOP=1 -d spaces_demo -c "alter database spaces_demo owner to spaces_migrador"

echo ok; sudo -u postgres psql -v ON_ERROR_STOP=1 -d spaces_demo <<'SQL'
do $$
declare r record;
begin
  for r in select tablename from pg_tables where schemaname='public' loop
    execute format('alter table public.%I owner to spaces_migrador', r.tablename);
  end loop;
  for r in select sequencename from pg_sequences where schemaname='public' loop
    execute format('alter sequence public.%I owner to spaces_migrador', r.sequencename);
  end loop;
  for r in select viewname from pg_views where schemaname='public' loop
    execute format('alter view public.%I owner to spaces_migrador', r.viewname);
  end loop;
end $$;
SQL
